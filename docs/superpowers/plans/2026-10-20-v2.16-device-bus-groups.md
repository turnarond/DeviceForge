# v2.16-B 设备栏标签分组 + 「打开这台设备」实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把横向胶囊设备栏改成「按标签分段 + 可折叠 + 可搜索」，并给每台设备一个「打开这台设备」的跳转入口（双击胶囊直达对应工具页），使 30–80 台规模下设备栏不再铺开不可用。

  ```cpp
  inline QStringList normalizedTags(const DeviceProfile& p)
  {
      QStringList out;
      for (const auto& tag : p.tags) {
          const QString normalized = normalizeTag(QString::fromStdString(tag));
          if (!normalized.isEmpty() && !out.contains(normalized)) out.append(normalized);
      }
      return out;
  }

  inline bool hasNormalizedTag(const DeviceProfile& p, const QString& selector)
  {
      const QString wanted = normalizeTag(selector);
      if (wanted.isEmpty()) return false;
      return normalizedTags(p).contains(wanted);
  }

  inline const QString& ungroupedTag()
  {
      static const QString kUngrouped = QStringLiteral("未分组");
      return kUngrouped;
  }

  inline bool matchesFilter(const DeviceProfile& p, const QString& filterText)
  {
      const QString needle = filterText.trimmed();
      if (needle.isEmpty()) return true;                    // 空过滤 = 全命中（规则 3 的前提）
      if (QString::fromStdString(p.name).contains(needle, Qt::CaseInsensitive)) return true;
      for (const auto& endpoint : p.endpoints)
          if (QString::fromStdString(endpoint.ip).contains(needle, Qt::CaseInsensitive)) return true;
      for (const auto& tag : normalizedTags(p))
          if (tag.contains(needle, Qt::CaseInsensitive)) return true;
      return false;
  }

  inline QList<Segment> segments(const std::vector<DeviceProfile>& profiles,
                                 const QSet<QString>& collapsedTags,
                                 const QString& filterText)
  {
      QMap<QString, QStringList> byTag;   // 标签 → 成员 deviceId（含「未分组」）
      for (const auto& profile : profiles) {
          const QString deviceId = QString::fromStdString(profile.deviceId);
          const QStringList tags = normalizedTags(profile);
          if (tags.isEmpty()) {
              byTag[ungroupedTag()].append(deviceId);
              continue;
          }
          for (const auto& tag : tags) byTag[tag].append(deviceId);
      }

      QStringList order;
      for (auto it = byTag.cbegin(); it != byTag.cend(); ++it)
          if (it.key() != ungroupedTag()) order.append(it.key());
      order.sort(Qt::CaseSensitive);           // 与 SylixOS/POSIX 目录口径一致
      if (byTag.contains(ungroupedTag())) order.append(ungroupedTag());

      const bool filtering = !filterText.trimmed().isEmpty();
      QList<Segment> out;
      for (const auto& tag : order) {
          Segment seg;
          seg.total = byTag.value(tag).size();
          for (const auto& deviceId : byTag.value(tag)) {
              const auto found = std::find_if(profiles.cbegin(), profiles.cend(),
                  [&deviceId](const DeviceProfile& p) {
                      return QString::fromStdString(p.deviceId) == deviceId;
                  });
              if (found == profiles.cend()) continue;
              if (filtering && !matchesFilter(*found, filterText)) continue;
              seg.deviceIds.append(deviceId);
          }
          if (!filtering) {
              seg.collapsed = collapsedTags.contains(tag);
          } else if (seg.deviceIds.isEmpty()) {
              seg.autoCollapsedBySearch = true;  // 未命中段整段折叠
          }
          out.append(seg);
      }
      return out;
  }
  ```

  ```cpp
  #pragma once
  #include <QString>
  #include "device/DeviceProfile.h"

  // v2.16-B：档案 → 目标工具页的唯一判定真源。Route 必须带命中端点的 ip——
  // 部署页设备下拉以纯 ip 为条目键（FtpDeployWidget.cpp:1544/1563），
  // 选中键与路由必须同源，否则会出现"路由选 A 端点、下拉选 B 端点"。
  namespace deviceopen {

  enum class Page { FtpDeploy, BatchCommand, Modbus, OpcUa, None };

  struct Route {
      Page page = Page::None;
      QString protocol;
      QString ip;
      int port = 0;
      QString statusHint;   // Page::None 时的状态栏文案（中文常量用 QStringLiteral）
  };

  inline Page pageFor(const QString& protocol)
  {
      if (protocol == QStringLiteral("ftp") || protocol == QStringLiteral("ftps")
          || protocol == QStringLiteral("sftp"))
          return Page::FtpDeploy;
      if (protocol == QStringLiteral("telnet") || protocol == QStringLiteral("ssh"))
          return Page::BatchCommand;
      if (protocol == QStringLiteral("modbus")) return Page::Modbus;
      if (protocol == QStringLiteral("opcua"))  return Page::OpcUa;
      return Page::None;
  }

  inline Route routeTo(const DeviceProfile& profile)
  {
      Route route;
      for (const auto& endpoint : profile.endpoints) {
          const QString ip = QString::fromStdString(endpoint.ip).trimmed();
          if (ip.isEmpty()) continue;                       // 空 ip 端点不可用，继续找下一个
          QString protocol = QString::fromStdString(endpoint.protocol).trimmed().toLower();
          if (protocol.isEmpty())
              protocol = QStringLiteral("ftp");             // 旧 device.list 行兜底
          const Page page = pageFor(protocol);
          if (page == Page::None) continue;                 // 协议不认识 ⇒ 不算命中
          route.page = page;
          route.protocol = protocol;
          route.ip = ip;
          route.port = endpoint.port;
          return route;
      }
      route.statusHint = QStringLiteral("该设备没有可用的协议端点");
      return route;
  }

  } // namespace deviceopen
  ```
