```cpp
#pragma once
#include "task/TaskTemplateTypes.h"
#include <QString>
#include <QStringList>
#include <QVariantMap>
#include <vector>
#include <string>

// v2.14 #49-2：单次部署配置 → 任务模板的纯映射（兑现 v2.11 spec §4「将当前单次
// 部署配置保存为新模板」承诺的可测性收敛层，参照 WsEventTypes.h 模式）。
// 只映射模板体系已能表达的字段：文件清单+目标目录+设备绑定+（可选）重启步骤；
// 部署前清空/FTPS/端口/并发度是工具级连接配置，刻意不进模板。
namespace deploymap {

struct OnceConfig { /* 见 Interfaces 块，字段逐一照抄 */ };

inline TaskTemplate buildTemplateFromOnceConfig(const OnceConfig& cfg)
{
    TaskTemplate tmpl;
    tmpl.name = QStringLiteral("单次部署模板").toStdString();   // 占位名，编辑器可改
    tmpl.version = 1;
    tmpl.deviceIds = cfg.deviceIds;
    TaskStep deploy;
    deploy.type = TaskStepType::DeployFiles;
    QVariantMap params;
    if (!cfg.protocol.isEmpty())
        params.insert(QStringLiteral("protocol"), cfg.protocol);
    params.insert(QStringLiteral("files"), cfg.localFiles);
    params.insert(QStringLiteral("remoteDirectory"),
                  cfg.remoteDirectory.isEmpty() ? QStringLiteral("/") : cfg.remoteDirectory);
    deploy.parameters = params;
    tmpl.steps.push_back(deploy);
    if (cfg.rebootAfter) {
        TaskStep reboot;
        reboot.type = TaskStepType::Reboot;
        QVariantMap rp;
        QString command = cfg.rebootCommand.trimmed();
        if (command.isEmpty()) command = QStringLiteral("reboot");   // 引擎缺省同口径
        rp.insert(QStringLiteral("command"), command);
        if (!cfg.rebootProtocol.isEmpty())
            rp.insert(QStringLiteral("protocol"), cfg.rebootProtocol);
        rp.insert(QStringLiteral("timeoutSec"), cfg.rebootTimeoutSec);
        rp.insert(QStringLiteral("retryCount"), cfg.rebootRetryCount);
        reboot.parameters = rp;
        tmpl.steps.push_back(reboot);
    }
    return tmpl;
}

} // namespace deploymap
```

void FtpDeployWidget::onSaveAsTemplateClicked()
{
    if (!m_deviceBus) { appendLog("设备总线未关联"); return; }
    // 与部署入口同源的拒绝分支（远程浏览源/空文件清单）
    if (m_leftPanel->source() && m_leftPanel->source()->sourceId() != "local") {
        appendLog("左侧面板当前为远程浏览源，请切换为「本地」后再保存模板");
        return;
    }
    const auto files = collectLocalFiles();
    if (files.empty()) { appendLog("请先在左侧本地面板中选中要部署的文件"); return; }

    deploymap::OnceConfig cfg;
    cfg.protocol = QString::fromStdString(currentProtocol());
    for (const auto& f : files) cfg.localFiles << QString::fromStdString(f);
    QString remoteDir = m_rightPanel ? m_rightPanel->currentPath() : QStringLiteral("/");
    if (remoteDir.isEmpty()) remoteDir = QStringLiteral("/");
    cfg.remoteDirectory = remoteDir;
    for (const auto& profile : m_deviceBus->selectedDeviceProfiles())
        cfg.deviceIds.push_back(profile.deviceId);   // 未勾选=留空（编辑器明示未绑定，不做"全部"展开）
    cfg.rebootAfter = m_rebootCheck->isChecked();
    cfg.rebootCommand = m_rebootCommandEdit->text();
    cfg.rebootProtocol = m_rebootProtocolCombo->currentData().toString();
    cfg.rebootTimeoutSec = m_rebootTimeoutSpin->value();
    cfg.rebootRetryCount = m_rebootRetrySpin->value();

    TaskTemplate initial = deploymap::buildTemplateFromOnceConfig(cfg);
    const std::vector<DeviceProfile> known = m_registry ? m_registry->list() : std::vector<DeviceProfile>{};
    TaskTemplateEditorDialog dlg(initial, known, this);
    if (dlg.exec() != QDialog::Accepted) return;
    TaskTemplate result;

```cpp
inline bool constantTimeEquals(const std::string& a, const std::string& b)
{
    const std::size_t n = std::max(a.size(), b.size());
    unsigned char accumulator = 0;
    for (std::size_t i = 0; i < n; ++i) {
        const unsigned char ca = i < a.size() ? static_cast<unsigned char>(a[i]) : 0u;
        const unsigned char cb = i < b.size() ? static_cast<unsigned char>(b[i]) : 0u;
        accumulator |= static_cast<unsigned char>(ca ^ cb);
        // volatile 落点：防 -O2 把累积或运算优化成短路比较（规格 §7 风险 3 处置）
        volatile unsigned char sink = accumulator;
        accumulator = sink;
    }
    accumulator |= static_cast<unsigned char>((a.size() ^ b.size()) != 0u);
    return accumulator == 0u;
}
```
